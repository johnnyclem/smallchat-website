import Foundation
import Combine

// MARK: - Domain Models

/// Represents a user in the chat system.
public struct User: Sendable, Codable, Identifiable, Hashable {
    public let id: String
    public let displayName: String
    public let avatarURL: URL?
    
    public init(id: String, displayName: String, avatarURL: URL? = nil) {
        self.id = id
        self.displayName = displayName
        self.avatarURL = avatarURL
    }
}

/// Represents a chat room or channel.
public struct Room: Sendable, Codable, Identifiable, Hashable {
    public let id: String
    public let name: String
    public let isPrivate: Bool
    
    public init(id: String, name: String, isPrivate: Bool = false) {
        self.id = id
        self.name = name
        self.isPrivate = isPrivate
    }
}

/// Represents a single message.
public struct Message: Sendable, Codable, Identifiable {
    public let id: String
    public let roomID: String
    public let author: User
    public let content: String
    public let timestamp: Date
    
    public init(id: String, roomID: String, author: User, content: String, timestamp: Date = .now) {
        self.id = id
        self.roomID = roomID
        self.author = author
        self.content = content
        self.timestamp = timestamp
    }
}

/// Connection state machine.
public enum ConnectionState: Sendable, Equatable {
    case disconnected
    case connecting
    case connected
    case failed(String)
    
    public static func == (lhs: ConnectionState, rhs: ConnectionState) -> Bool {
        switch (lhs, rhs) {
        case (.disconnected, .disconnected),
             (.connecting, .connecting),
             (.connected, .connected):
            return true
        case (.failed(let l), .failed(let r)):
            return l == r
        default:
            return false
        }
    }
}

// MARK: - Transport & API Protocols

public protocol ChatTransport: Sendable {
    var receivedMessages: AsyncThrowingStream<Data, any Error> { get }
    func connect() async throws
    func send(data: Data) async throws
    func disconnect() async
}

public protocol ChatAPI: Sendable {
    func fetchRooms() async throws -> [Room]
    func fetchMessageHistory(for roomID: String, limit: Int) async throws -> [Message]
}

// MARK: - Event Coding (Completed)

private enum EventType: String, Codable {
    case message = "chat.message"
    case userJoined = "chat.user_joined"
    case error = "chat.error"
}

private struct OutgoingEvent: Encodable {
    let type: EventType
    let payload: Payload
    
    enum Payload: Encodable {
        case message(roomID: String, content: String)
        
        func encode(to encoder: Encoder) throws {
            var container = encoder.container(keyedBy: CodingKeys.self)
            switch self {
            case .message(let roomID, let content):
                try container.encode(roomID, forKey: .roomID)
                try container.encode(content, forKey: .content)
            }
        }
        
        enum CodingKeys: String, CodingKey {
            case roomID = "room_id"
            case content
        }
    }
    
    static func sendMessage(roomID: String, content: String) -> OutgoingEvent {
        return OutgoingEvent(type: .message, payload: .message(roomID: roomID, content: content))
    }
}

private enum IncomingEvent: Decodable {
    case newMessage(Message)
    case userJoined(user: User, roomID: String)
    case error(String)
    
    enum CodingKeys: String, CodingKey {
        case type
        case payload
    }
    
    init(from decoder: Decoder) throws {
        let container = try decoder.container(keyedBy: CodingKeys.self)
        let type = try container.decode(EventType.self, forKey: .type)
        
        switch type {
        case .message:
            let message = try container.decode(Message.self, forKey: .payload)
            self = .newMessage(message)
        case .userJoined:
            let payload = try container.decode(JoinPayload.self, forKey: .payload)
            self = .userJoined(user: payload.user, roomID: payload.roomID)
        case .error:
            let message = try container.decode(String.self, forKey: .payload)
            self = .error(message)
        }
    }
    
    struct JoinPayload: Decodable {
        let user: User
        let roomID: String
        enum CodingKeys: String, CodingKey {
            case user
            case roomID = "room_id"
        }
    }
}

// MARK: - Chat Client Actor

public actor ChatClient {
    
    public struct Configuration: Sendable {
        public let apiBaseURL: URL
        public let wsBaseURL: URL
        public let heartbeatInterval: Duration
        
        public init(apiBaseURL: URL, wsBaseURL: URL, heartbeatInterval: Duration = .seconds(30)) {
            self.apiBaseURL = apiBaseURL
            self.wsBaseURL = wsBaseURL
            self.heartbeatInterval = heartbeatInterval
        }
    }
    
    private let transport: ChatTransport
    private let api: ChatAPI
    private let config: Configuration
    
    private(set) public var currentUser: User?
    private(set) public var connectionState: ConnectionState = .disconnected
    
    private var rooms: [String: Room] = [:]
    private var messages: [String: [Message]] = [:]
    
    private var messageContinuation: AsyncStream<Message>.Continuation?
    private var stateContinuation: AsyncStream<ConnectionState>.Continuation?
    
    public let messageStream: AsyncStream<Message>
    public let stateStream: AsyncStream<ConnectionState>
    
    public init(transport: ChatTransport, api: ChatAPI, configuration: Configuration) {
        self.transport = transport
        self.api = api
        self.config = configuration
        
        let (mStream, mCont) = AsyncStream<Message>.makeStream()
        let (sStream, sCont) = AsyncStream<ConnectionState>.makeStream()
        
        self.messageContinuation = mCont
        self.stateContinuation = sCont
        self.messageStream = mStream
        self.stateStream = sStream
    }
    
    deinit {
        messageContinuation?.finish()
        stateContinuation?.finish()
    }
    
    public func login(user: User) async {
        self.currentUser = user
        await connect()
    }
    
    public func connect() async {
        guard connectionState != .connected, connectionState != .connecting else { return }
        await updateState(.connecting)
        
        do {
            try await transport.connect()
            await updateState(.connected)
            
            Task { [weak self] in
                guard let self else { return }
                await self.listenForTransportMessages()
            }
        } catch {
            await updateState(.failed(error.localizedDescription))
        }
    }
    
    public func sendMessage(to roomID: String, content: String) async throws {
        guard let user = currentUser else { throw ChatError.notAuthenticated }
        
        // Optimistic update
        let optimisticMessage = Message(
            id: UUID().uuidString,
            roomID: roomID,
            author: user,
            content: content
        )
        appendMessage(optimisticMessage, to: roomID)
        messageContinuation?.yield(optimisticMessage)
        
        let event = OutgoingEvent.sendMessage(roomID: roomID, content: content)
        let payload = try JSONEncoder().encode(event)
        try await transport.send(data: payload)
    }
    
    public func bootstrap() async throws {
        let fetchedRooms = try await api.fetchRooms()
        for room in fetchedRooms {
            self.rooms[room.id] = room
            // Fetch history per room concurrently
            Task {
                let history = try? await api.fetchMessageHistory(for: room.id, limit: 50)
                if let history {
                    await self.setHistory(history, for: room.id)
                }
            }
        }
    }
    
    private func setHistory(_ history: [Message], for roomID: String) {
        messages[roomID] = history
    }
    
    public func getRooms() -> [Room] {
        Array(rooms.values)
    }
    
    public func getMessages(for roomID: String) -> [Message] {
        messages[roomID] ?? []
    }
    
    private func listenForTransportMessages() async {
        do {
            for try await data in transport.receivedMessages {
                let event = try JSONDecoder().decode(IncomingEvent.self, from: data)
                await handleEvent(event)
            }
        } catch {
            await updateState(.failed(error.localizedDescription))
        }
    }
    
    private func handleEvent(_ event: IncomingEvent) async {
        switch event {
        case .newMessage(let message):
            appendMessage(message, to: message.roomID)
            messageContinuation?.yield(message)
        case .userJoined(let user, let roomID):
            print("User \(user.displayName) joined \(roomID)")
        case .error(let msg):
            print("Server Error: \(msg)")
        }
    }
    
    private func appendMessage(_ message: Message, to roomID: String) {
        if messages[roomID] == nil { messages[roomID] = [] }
        // Prevent duplicates from optimistic UI vs server confirmation
        if !(messages[roomID]?.contains(where: { $0.id == message.id }) ?? false) {
            messages[roomID]?.append(message)
        }
    }
    
    private func updateState(_ newState: ConnectionState) async {
        self.connectionState = newState
        stateContinuation?.yield(newState)
    }
}

// MARK: - Implementations

public final class WebSocketTransport: ChatTransport, @unchecked Sendable {
    private let url: URL
    private var webSocketTask: URLSessionWebSocketTask?
    private let session = URLSession(configuration: .default)
    
    private var continuation: AsyncThrowingStream<Data, any Error>.Continuation?
    
    public let receivedMessages: AsyncThrowingStream<Data, any Error>
    
    public init(url: URL) {
        self.url = url
        let (stream, cont) = AsyncThrowingStream<Data, any Error>.makeStream()
        self.receivedMessages = stream
        self.continuation = cont
    }
    
    public func connect() async throws {
        webSocketTask = session.webSocketTask(with: url)
        webSocketTask?.resume()
        startListening()
    }
    
    private func startListening() {
        webSocketTask?.receive { [weak self] result in
            switch result {
            case .success(let message):
                switch message {
                case .data(let data):
                    self?.continuation?.yield(data)
                case .string(let text):
                    if let data = text.data(using: .utf8) {
                        self?.continuation?.yield(data)
                    }
                @unknown default:
                    break
                }
                self?.startListening()
            case .failure(let error):
                self?.continuation?.finish(throwing: error)
            }
        }
    }
    
    public func send(data: Data) async throws {
        try await webSocketTask?.send(.data(data))
    }
    
    public func disconnect() async {
        webSocketTask?.cancel(with: .normalClosure, reason: nil)
        continuation?.finish()
    }
}

public final class ChatAPIClient: ChatAPI, Sendable {
    private let baseURL: URL
    
    public init(base: URL) {
        self.baseURL = base
    }
    
    public func fetchRooms() async throws -> [Room] {
        let url = baseURL.appendingPathComponent("rooms")
        let (data, _) = try await URLSession.shared.data(from: url)
        return try JSONDecoder().decode([Room].self, from: data)
    }
    
    public func fetchMessageHistory(for roomID: String, limit: Int) async throws -> [Message] {
        let url = baseURL.appendingPathComponent("rooms/\(roomID)/messages")
        // Implementation of query params omitted for brevity
        let (data, _) = try await URLSession.shared.data(from: url)
        return try JSONDecoder().decode([Message].self, from: data)
    }
}
